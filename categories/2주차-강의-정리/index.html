[이어드림스쿨6기_2주차] 0629 Pandas 라이브러리 복습

Pandas 행과 열로 이루어진 데이터를 만들고 다루는 패키지 데이터 분석에 특화 => 대용량 데이터 처리에 편리 자료 구조와 데이터 조작 기능 제공 import pandas as pd Series 1차원 배열형 구조 인덱스를 포함 (기본 인덱스는 0부터 시작) index 매개변수로 사용자 인덱스 지정 가능 인덱스 순서가 다른 Series끼리 연산 시 인덱스 라벨로 맞춤 # 기본 Series — 인덱스 0부터 자동 obj = pd.Series([4, 7, -5]) print("Basic Series:") print(obj) print() # 사용자 지정 인덱스 Series (날짜 문자열 등) obj2 = pd.Series([4, 7, -5], index=['2016-01-01', '2016-01-02', '2016-01-03']) print("Series with custom index:") print(obj2) print() # 인덱스 라벨로 원소 접근 print("Value at '2016-01-01':", obj2['2016-01-01']) print() # Series 연산 — 위치가 아닌 인덱스 라벨로 맞춤 stock0 = pd.Series([10, 20, 30], index=['naver', 'skt', 'LG']) stock1 = pd.Series([30, 20, 10], index=['LG', 'naver', 'skt']) print("stock0:") print(stock0) print("stock1 (different order):") print(stock1) # 순서가 달라도 라벨 기준으로 덧셈 merge = stock0 + stock1 print("stock0 + stock1 (matched by label):") print(merge) # 결과 Basic Series: 0 4 1 7 2 -5 dtype: int64 Series with custom index: 2016-01-01 4 2016-01-02 7 2016-01-03 -5 dtype: int64 Value at '2016-01-01': 4 stock0: naver 10 skt 20 LG 30 dtype: int64 stock1 (different order): LG 30 naver 20 skt 10 dtype: int64 stock0 + stock1 (matched by label): LG 60 naver 30 skt 30 dtype: int64 DataFrame Series가 모여 행과 열을 이룬 데이터 (2차원 표 구조) Dataframe 생성에는 딕셔너리 사용 키 → 열 이름 값(리스트) → 열 데이터 열 방향: 키-값 쌍 하나가 한 열 행 방향: 정수 인덱스 자동 부여(Series와 유사), 또는 사용자 인덱스 지정 from pandas import Series, DataFrame # 딕셔너리로 DataFrame 생성 raw_data = { 'col0' : [1, 2, 3, 4], 'col1' : [10, 20, 30, 40], 'col2' : [100, 200, 300, 400] } data = DataFrame(raw_data) print("DataFrame from dictionary:") print(data) print() # 2D 리스트 + 사용자 index·columns a = [[1, 2], [3, 4]] df = DataFrame(a, index=[0, 1], columns=['a', 'b']) print("DataFrame from 2D list:") print(df) print() # 사용자 행 인덱스·열 순서 지정 daeshin = { 'open': [11650, 11100, 11200, 11100, 11000], 'high': [12100, 11800, 11200, 11100, 11150], 'low': [11600, 11050, 10900, 10950, 10900], 'close': [11900, 11600, 11000, 11100, 11050] } daeshin_day = DataFrame(daeshin, index=[0, 2, 4, 6, 8], columns=['open', 'high', 'low', 'close']) print("Stock price DataFrame with custom index:") print(daeshin_day) # 결과 DataFrame from dictionary: col0 col1 col2 0 1 10 100 1 2 20 200 2 3 30 300 3 4 40 400 DataFrame from 2D list: a b 0 1 2 1 3 4 Stock price DataFrame with custom index: open high low close 0 11650 12100 11600 11900 2 11100 11800 11050 11600 4 11200 11200 10900 11000 6 11100 11100 10950 11100 8 11000 11150 10900 11050 인덱싱 & 슬라이싱 .loc[] : 라벨(이름) 기반 인덱싱. 끝 포함 .iloc[] : 정수 위치 기반 인덱싱. 끝 미포함 # 문자열 인덱스 샘플 DataFrame data = {'A': [1, 2, 3], 'B': [4, 5, 6], 'C': [7, 8, 9]} df = pd.DataFrame(data, index=['a', 'b', 'c']) print("DataFrame:") print(df) print() # loc — 라벨: 행 'a'~'b', 열 'A'·'C' # 참고: 끝 라벨 'b' 포함 print("df.loc['a':'b', ['A','C']]:") print(df.loc['a':'b', ['A', 'C']]) print() # iloc — 정수: 행 0~1(2 미포함), 열 0·2 # 참고: 인덱스 2 미포함 (끝 제외) print("df.iloc[0:2, [0,2]]:") print(df.iloc[0:2, [0, 2]]) # 결과 DataFrame: A B C a 1 4 7 b 2 5 8 c 3 6 9 df.loc['a':'b', ['A','C']]: A C a 1 7 b 2 8 df.iloc[0:2, [0,2]]: A C a 1 7 b 2 8 데이터 삭제 .drop() : DataFrame에서 행 또는 열을 삭제 # 샘플 DataFrame data = {'A': [1, 2, 3], 'B': [4, 5, 6], 'C': [7, 8, 9]} df = pd.DataFrame(data, index=['a', 'b', 'c']) print("Original DataFrame:") print(df) print() # 행 'a' 삭제 (axis=0, 기본값) df_dropped_row = df.drop(labels='a') print("After dropping row 'a':") print(df_dropped_row) print() # 열 'A' 삭제 (axis=1 지정) df_dropped_col = df.drop(labels='A', axis=1) print("After dropping column 'A':") print(df_dropped_col) print() # inplace=False (기본값, 원본 df 유지) # inplace=True (원본 df 변경) # ==> 데이터 가공 후, 원본 업데이트 됨 # ==> 비추. 코딩 일관성 측면에서 좋지 않음 # df2 = df.가공 (df 가공하여 df2에 저장) # ==> 일반적으로 위 방법을 통하여 진행 # 결과 Original DataFrame: A B C a 1 4 7 b 2 5 8 c 3 6 9 After dropping row 'a': A B C b 2 5 8 c 3 6 9 After dropping column 'A': B C a 4 7 b 5 8 c 6 9 데이터 정렬 .sort_values() : 열 값 기준으로 DataFrame 또는 Series 정렬 # 샘플 DataFrame data = {'A': [3, 2, 1], 'B': [4, 5, 6]} df = pd.DataFrame(data) print("Original DataFrame:") print(df) print() # 열 'A' 오름차순 정렬 (기본) sorted_df = df.sort_values(by='A') print("Sorted by column 'A' (ascending):") print(sorted_df) print("Note: the index order also changes (2, 1, 0)") print() # 내림차순 정렬 sorted_desc = df.sort_values(by='A', ascending=False) print("Sorted by column 'A' (descending):") print(sorted_desc) # 결과 Original DataFrame: A B 0 3 4 1 2 5 2 1 6 Sorted by column 'A' (ascending): A B 2 1 6 1 2 5 0 3 4 Note: the index order also changes (2, 1, 0) Sorted by column 'A' (descending): A B 0 3 4 1 2 5 2 1 6 데이터프레임 간의 연산 .add() # 덧셈 .sub() # 뺄셈 .mul() # 곱셈 .div() # 나눗셈 # 연산 시연용 DataFrame 두 개 df1 = pd.DataFrame({'A': [1, 2, 3], 'B': [4, 5, 6]}) df2 = pd.DataFrame({'A': [10, 20, 30], 'B': [40, 50, 60]}) print("df1:") print(df1) print("df2:") print(df2) print() # 덧셈 print("df1.add(df2):") print(df1.add(df2)) print() # 뺄셈 print("df1.sub(df2):") print(df1.sub(df2)) print() # fill_value — NaN 처리 예제 df3 = pd.DataFrame({'A': [1, 2], 'B': [3, 4]}) df4 = pd.DataFrame({'A': [10, 20, 30], 'B': [40, 50, 60]}) print("df3 (2 rows) + df4 (3 rows) without fill_value:") print(df3.add(df4)) print() print("df3 + df4 with fill_value=0 (NaN replaced by 0):") print(df3.add(df4, fill_value=0)) #
